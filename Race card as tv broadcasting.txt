<script>
  const runners = [
    {num:1, name:"DAIANNE", trainer:"Bhawani Singh"},
    {num:2, name:"LOOKING LIKE A WOW", trainer:"Ramswarup"},
    {num:3, name:"SILVER BRAID", trainer:"Aditya Waydande"},

    {num:5, name:"HIGHGROUND", trainer:"S J Sunil"},
    {num:6, name:"LEO THE LION", trainer:"N Bhosale"}
  ];

  const scroll = document.getElementById("runnerScroll");
  function addRunners() {
    runners.forEach(r => {
      const box = document.createElement("div");
      box.className = "runner-box";
      box.innerHTML = `
        <div class="runner-number">${r.num}</div>
        <div class="runner-name">${r.name}</div>
        <div class="trainer">Jockey: ${r.trainer}</div>
      `;
      scroll.appendChild(box);
    });
  }

  // Repeat runners twice for seamless scroll
  addRunners();
  addRunners();
</script>
